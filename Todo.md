import Image from "next/image";
import { prisma } from "@/lib/prisma";
import { cookies } from "next/headers";
import { FileText } from "lucide-react";

export default async function CertificatePage() {
  
  const cookieStore = await cookies();
  const studentId = cookieStore.get("student_session")?.value;

  if (!studentId) {
  throw new Error("Not authenticated");
}

  const certificate =
    await prisma.clearanceCertificate.findUnique({
      where: {
        studentId,
      },
      include: {
        student: true,
      },
    });

  if (!certificate) {
    return (
      <div className="flex min-h-screen items-center justify-center">
        <div className="rounded-xl border bg-white p-8 shadow">
          <h2 className="text-2xl font-bold">
            Certificate Not Available
          </h2>

          <p className="mt-3 text-gray-500">
            Complete all clearance requirements to
            receive your certificate.
          </p>
        </div>
      </div>
    );
  }

  return (
   <main className="min-h-screen bg-slate-100 py-12">
  <div
    id="certificate"
    className="mx-auto w-full max-w-4xl rounded-xl border-4 border-green-700 bg-white p-6 md:p-10 shadow-xl"
  >
    {/* Header */}
    <div className="flex flex-col gap-6 md:flex-row md:items-center md:justify-between border-b pb-6">
      <Image
        src="/unnlogo.png"
        width={90}
        height={90}
        alt="UNN Logo"
      />

      <div className="text-center">
        <h1 className="text-2xl md:text-3xl font-bold uppercase">
          University Clearance Certificate
        </h1>

        <p className="text-gray-500 mt-2">
          Generated by SmartClear Automated Clearance Management System
        </p>
      </div>

      <div className="text-right">
        <div className="flex items-center gap-2 justify-end">
          <FileText className="text-green-700" />
          <span className="font-bold text-xl">
            SmartClear
          </span>
        </div>
      </div>
    </div>

    {/* Body */}
    <div className="mt-8 text-center">
      <p className="mx-auto mt-6 max-w-2xl text-base md:text-lg leading-8 text-gray-700">
        THIS CERTIFIES THAT
      </p>

      <h2 className="mt-5 text-3xl md:text-5xl font-bold tracking-wide uppercase text-green-700">
        {certificate.student.fullName}
      </h2>

      <p className="mt-8 text-lg leading-9 text-gray-700 max-w-3xl mx-auto">
        has successfully completed all required graduation clearance
        procedures of the University and has satisfied every
        administrative requirement necessary for graduation.
      </p>
    </div>

    {/* Details */}
    <div className="mt-10 grid gap-10 md:grid-cols-2">
      <div className="grid grid-cols-2 gap-x-6 gap-y-5">
        <div>
          <p className="text-gray-500 text-sm">
            Registration Number
          </p>
          <p className="font-semibold text-lg">
            {certificate.student.regNo}
          </p>
        </div>

        <div>
          <p className="text-gray-500 text-sm">
            Department
          </p>
          <p className="font-semibold text-lg">
            {certificate.student.department}
          </p>
        </div>

        <div>
          <p className="text-gray-500 text-sm">
            Level
          </p>
          <p className="font-semibold text-lg">
            {certificate.student.level}
          </p>
        </div>

        <div>
          <p className="text-gray-500 text-sm">
            Date Issued
          </p>
          <p className="font-semibold">
            {certificate.issuedAt.toLocaleDateString()}
          </p>
        </div>

        <div>
          <p className="text-gray-500 text-sm">
            Verification Code
          </p>

          <p className="font-mono font-bold text-lg text-green-700">
            {certificate.verificationCode}
          </p>
        </div>
      </div>

      <div className="flex flex-col items-center">
        <div className="rounded-lg border bg-white p-3 shadow-sm">
          <Image
            src={certificate.qrCodeUrl!}
            alt="QR Code"
            width={220}
            height={220}
          />
        </div>

        <p className="mt-5 text-sm text-center text-gray-500">
          Scan this QR code to verify this certificate.
        </p>
      </div>
    </div>

    {/* Signature */}
    <div className="mt-20 flex justify-between items-end">
      <div>
        <div className="border-t border-black w-64"></div>

        <p className="mt-2 font-semibold">
          Registrar / Clearance Officer
        </p>
      </div>

      <div className="text-right text-sm text-gray-500">
        Verify online using the QR Code
        <br />
        or the verification code above.
      </div>
    </div>
  </div>
</main>
  );
}
